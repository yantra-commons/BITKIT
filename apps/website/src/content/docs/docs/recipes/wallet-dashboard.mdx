---
title: Wallet dashboard
description: Balance, send form with fee selection, and transaction history.
---

The kit gives you the screens. Your wallet backend (a node, an SDK, a watch-only xpub service)
provides the data.

```html
<ou-balance-card id="balance" fiat-currency="USD" fiat-price="97250">
  <div slot="actions">
    <ou-button id="send">Send</ou-button><ou-button variant="secondary">Receive</ou-button>
  </div>
</ou-balance-card>

<form id="send-form">
  <ou-address-input name="to" label="To" accept="onchain-address,bip21" required></ou-address-input>
  <ou-amount-input name="amount" label="Amount" required></ou-amount-input>
  <ou-fee-selector id="fees" vsize="141"></ou-fee-selector>
  <ou-button type="submit" block>Review transaction</ou-button>
</form>

<ou-transaction-list id="history" interactive has-more></ou-transaction-list>
```

```ts
const wallet = await myWalletBackend();

balance.balance = wallet.confirmedSats; // bigint
balance.pending = wallet.unconfirmedSats;
document.querySelector('ou-amount-input').balance = wallet.confirmedSats;
history.transactions = await wallet.transactions(); // TransactionData[]

// Inputs are form-associated: FormData just works.
document.querySelector('#send-form').addEventListener('submit', (e) => {
  e.preventDefault();
  const data = new FormData(e.target); // { to: 'bc1…', amount: '21000' }
  const fee = document.querySelector('#fees').selection; // { tier, satPerVb, feeSats }
});

// When paste or scan yields a BIP21 link with an amount, prefill it.
document.querySelector('ou-address-input').addEventListener('ou-change', (e) => {
  const r = e.detail.result;
  if (r?.type === 'bip21' && r.uri.amountMsat)
    document.querySelector('ou-amount-input').amount = r.uri.amountMsat / 1000n;
});
```

For coin control, add `<ou-utxo-list>` and set `target` to the amount plus fee. See the full
[wallet dashboard template](../../../templates/#wallet-dashboard).
